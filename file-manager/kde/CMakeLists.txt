cmake_minimum_required(VERSION 3.16)
project(neutrondrive-overlay LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(NEUTRONDRIVE_KF_MAJOR "6" CACHE STRING "Match Dolphin's KDE Frameworks major (5 or 6)")
if(NEUTRONDRIVE_KF_MAJOR STREQUAL "6")
    find_package(KF6KIO REQUIRED)
elseif(NOT NEUTRONDRIVE_KF_MAJOR STREQUAL "5")
    message(FATAL_ERROR "NEUTRONDRIVE_KF_MAJOR must be 5 or 6")
endif()
if(NEUTRONDRIVE_KF_MAJOR STREQUAL "6")
    set(ND_KIO KF6::KIOCore KF6::KIOFileWidgets)
    set(ND_NAMESPACE kf6/overlayicon)
else()
    find_package(KF5KIO REQUIRED)
    set(ND_KIO KF5::KIOCore KF5::KIOFileWidgets)
    set(ND_NAMESPACE kf5/overlayicon)
endif()
if(NEUTRONDRIVE_KF_MAJOR STREQUAL "6")
    include(KDEInstallDirs6)
else()
    include(KDEInstallDirs5)
endif()
add_library(neutrondriveoverlay MODULE overlay.cpp)
target_link_libraries(neutrondriveoverlay PRIVATE ${ND_KIO})
install(TARGETS neutrondriveoverlay DESTINATION ${KDE_INSTALL_PLUGINDIR}/${ND_NAMESPACE})

# Tests may point at a temporary Rust library; host packages use the default.
if(DEFINED NEUTRONDRIVE_RUST_LIBRARY)
    target_compile_definitions(neutrondriveoverlay PRIVATE NEUTRONDRIVE_RUST_LIBRARY="${NEUTRONDRIVE_RUST_LIBRARY}")
endif()
